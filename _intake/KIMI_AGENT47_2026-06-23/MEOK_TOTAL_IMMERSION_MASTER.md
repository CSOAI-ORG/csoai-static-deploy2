### Code: Connect a Unitree Go2 to CSOAI Agents

```python
# unitree_csoai_bridge.py — connect physical robot to MEOK agents
import asyncio
from unitree_go2 import Go2Robot  # pip install unitree_sdk2
from csoai_agent import AgentTown  # Your agent system

class RobotAgentBridge:
    """A Unitree Go2 robot becomes Agent #48 in MEOK town"""
    
    def __init__(self):
        self.robot = Go2Robot()  # Connect to physical robot
        self.agent = AgentTown.spawn_agent(role="patrol")  # Spawn in MEOK
        self.pheromone_trail = []  # Track where robot goes
    
    async def patrol_loop(self):
        while True:
            # Robot senses environment (cameras, LiDAR)
            observation = self.robot.get_observation()
            
            # Agent decides where to go next
            action = self.agent.decide(observation)
            
            # Robot moves physically
            self.robot.move(action)
            
            # Agent emits pheromone signal
            self.agent.emit_pheromone("PATROL", location=action.position)
            
            # Record trail for visualization
            self.pheromone_trail.append({
                "agent": self.agent.id,
                "location": action.position,
                "signal": "PATROL",
                "timestamp": time.now()
            })
            
            await asyncio.sleep(0.1)  # 10Hz control loop

# Run: python unitree_csoai_bridge.py
# Result: Physical robot patrolling MEOK town, leaving pheromone trails
```
